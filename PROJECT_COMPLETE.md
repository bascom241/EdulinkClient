## 💡 USAGE EXAMPLES

### Using Components
```tsx
import { Button, Card, Modal } from '@/components/ui'

// Button
<Button variant="primary" onClick={handleClick}>
  Click me
</Button>

// Card
<Card>
  <CardContent>Your content</CardContent>
</Card>

// Modal
<Modal isOpen={open} onClose={handleClose} title="Title">
  Modal content
</Modal>
```
